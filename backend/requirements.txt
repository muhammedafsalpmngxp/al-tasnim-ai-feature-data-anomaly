# Pinned to exact versions. Bump deliberately, one package at a time, then re-run
# `python -m eval.run_eval` before committing an upgrade — an unbounded `>=` lets a routine
# `pip install` silently jump a major version.
#
# Install into your conda environment:
#     conda activate <your-env>
#     pip install -r requirements.txt

# ── Core orchestration ──────────────────────────────────────────────────────────
